\noindent The component form of the product \(\pv{F} \pv{U}/\gamma\) used in the Lorentz-force law is given in Table \ref{tab:components-fu}.
\begin{componenttable}[tab:components-fu]{\pv{F} \pv{U}/\gamma}
\componentrow{real scalar}{\vct{E}\cdot\vct{v}}{}
\componentrow{img scalar}{\vct{B}\cdot\vct{v}}{}
\componentrow{real vector}{\vct{E}+\vct{v}\times\vct{B}}{}
\componentrow{img vector}{\vct{B}+\vct{E}\times\vct{v}}{}
\end{componenttable}
