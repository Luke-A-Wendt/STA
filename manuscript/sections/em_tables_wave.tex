\noindent The scalar and vector parts of Maxwell's equation and its sourced wave consequence are recorded in Tables \ref{tab:components-maxwell} and \ref{tab:components-wave-f}.
\begin{componenttable}[tab:components-maxwell]{\pv{\partial} \pv{F}=\rho-\vct{J}\cdot\sigv}
\componentrow{real scalar}{\gradv\cdot\vct{E}=\rho}{}
\componentrow{img scalar}{\gradv\cdot\vct{B}=0}{}
\componentrow{real vector}{\partial_t\vct{E}-\gradv\times\vct{B}=-\vct{J}}{}
\componentrow{img vector}{\partial_t\vct{B}+\gradv\times\vct{E}=0}{}
\end{componenttable}

\begin{componenttable}[tab:components-wave-f]{\Box \pv{F}=\pv{\partial}^{*}(\rho-\vct{J}\cdot\sigv)}
\componentrow{real scalar}{\partial_t\rho+\gradv\cdot\vct{J}=0}{}
\componentrow{img scalar}{0=0}{}
\componentrow{real vector}{\Box\vct{E}=-\partial_t\vct{J}-\gradv\rho}{}
\componentrow{img vector}{\Box\vct{B}=\gradv\times\vct{J}}{}
\end{componenttable}
