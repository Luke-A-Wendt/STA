\subsubsection{Functions of \texorpdfstring{\(\vct{u}\cdot\sigv\)}{u.sigma}}
The two-root decomposition of \(\vct{u}\cdot\sigv\), whose two values are
\(+1\) and \(-1\), is given by the projector pair and identities in
Eq.\ \eqref{eq:projectors} together with the functional-calculus rule in
Eq.\ \eqref{eq:functional-calculus}. The function is determined by
specified scalar values at those roots:
\begin{align*}
f(\vct{u}\cdot\sigv)
&= f(+)\pv{\Pi}(+) + f(-)\pv{\Pi}(-)\notag \\
&= \tfrac{1}{2}\bigl(f(+)+f(-)\bigr)
+ \tfrac{1}{2}\bigl(f(+)-f(-)\bigr)\,\vct{u}\cdot\sigv.
\end{align*}
These are the basic projector formulas for a unit-direction element. The
square root, logarithm branch data, rational inverse, Cayley transform, and
the two exponential families use the branch-sign function defined by
\begin{equation}
\epsilon(\pm):=\eqtext{an independently chosen member of }\{+1,-1\}.
\label{eq:branch-sign-function}
\end{equation}
The resulting identities are
\begin{align*}
\sqrt{\vct{u}\cdot\sigv}
&= \epsilon(+)\pv{\Pi}(+) + \epsilon(-) i\,\pv{\Pi}(-), \\
\log(\vct{u}\cdot\sigv)
&= 2 i \pi\bigl(n\,\pv{\Pi}(+) + (\tfrac12+m)\,\pv{\Pi}(-)\bigr), \\
(z+\vct{u}\cdot\sigv)^{-1}
&= \frac{1}{z+1}\pv{\Pi}(+) + \frac{1}{z-1}\pv{\Pi}(-)\notag \\
&= \frac{z-\vct{u}\cdot\sigv}{z^{2}-1}, \\
\frac{i-\vct{u}\cdot\sigv}{i+\vct{u}\cdot\sigv}
&= i\,\vct{u}\cdot\sigv, \\
\exp(\theta\,\vct{u}\cdot\sigv)
&= \cosh(\theta)+\sinh(\theta)\,\vct{u}\cdot\sigv, \\
\exp(i\theta\,\vct{u}\cdot\sigv)
&= \cos(\theta)+i\sin(\theta)\,\vct{u}\cdot\sigv,
\end{align*}
where \(m\) and \(n\) are integers indexing independent logarithm branches,
and the
inverse formula requires \(z\neq\pm1\). The two values from
Eq.\ \eqref{eq:branch-sign-function} are independent; the
conventional scalar branches \(\sqrt{+1}=1\) and \(\sqrt{-1}=i\) are obtained
when \(\epsilon(+)=\epsilon(-)=+1\). These identities are obtained by applying
the chosen scalar value separately on the two projector ranges.
The corresponding domains for real and complex paravectors are stated with
the exponential and logarithm formulas.

\subsubsection{Functions of \pv{X}}
The scalar, vector, and unit-direction data are specified by
\begin{align*}
\pv{X} &= t + \vct{r}\cdot\sigv, \\
t &= \eqtext{scalar}(\pv{X})\in\R, \\
\vct{r} &= \eqtext{vector}(\pv{X})\in\R^{3}, \\
\vct{u} &:= \vct{r}/\lVert\vct{r}\rVert \qquad \text{when }\vct{r}\neq 0.
\end{align*}
If \(\vct{r}\neq 0\), its decomposition is
\(\vct{r}=\lVert\vct{r}\rVert \vct{u}\), with \(\vct{u}^{2}=1\).
The two eigenvalue values are defined by
\begin{equation}
\lambda(\pv{X},\pm):=t\pm\lVert\vct{r}\rVert.
\label{eq:real-paravector-eigenvalues}
\end{equation}
With the eigenvalue function fixed by
Eq.\ \eqref{eq:real-paravector-eigenvalues}, the minimal polynomial of
\(\pv{X}\) is
\begin{align*}
\pv{X}^{2} - 2 t \pv{X} + ( t^{2} - \vct{r}^{2} ) &= 0,\notag \\
0 &= (\pv{X}-\lambda(\pv{X},-))(\pv{X}-\lambda(\pv{X},+))\notag \\
&= \pv{X}^{2} - \lambda(\pv{X},-)\pv{X}\notag \\
&\quad - \pv{X}\lambda(\pv{X},+) + \lambda(\pv{X},-)\lambda(\pv{X},+).
\end{align*}
With
\begin{equation*}
a:=\lambda(\pv{X},+),
\qquad
b:=\lambda(\pv{X},-),
\end{equation*}
the projector function from Eq.\ \eqref{eq:projectors} gives the
corresponding decomposition:
\begin{align*}
\pv{\Pi}(\pm) &= \tfrac12(1\pm \vct{u}\cdot\sigv), \\
\pv{X} &= a\,\pv{\Pi}(+) + b\,\pv{\Pi}(-), \\
\pv{\Pi}(\pm)\pv{X} &= \lambda(\pv{X},\pm)\pv{\Pi}(\pm).
\end{align*}
If \(f\) is defined at the two eigenvalues, then
\begin{align*}
f(\pv{X})
&= f(a)\pv{\Pi}(+) + f(b)\pv{\Pi}(-)\notag \\
&= \tfrac12\bigl(f(a)+f(b)\bigr)
+ \tfrac12\bigl(f(a)-f(b)\bigr)\,\vct{u}\cdot\sigv.
\end{align*}
The scalar and signed axial coefficients of \(f(\pv{X})\) are defined by
\begin{align*}
t' &= \tfrac12\bigl(f(a)+f(b)\bigr), \\
\rho'&:=\tfrac12\bigl(f(a)-f(b)\bigr),\\
f(\pv{X})&=t'+\rho'\,\vct{u}\cdot\sigv.
\end{align*}
For complex-valued \(f\), both \(t'\) and \(\rho'\) may be complex. Even
when they are real, \(\rho'\) is signed; the magnitude of the vector part is
\(\lvert\rho'\rvert\), not \(\rho'\) itself.
The projector summary obtained from
Eqs.\ \eqref{eq:real-paravector-eigenvalues} and \eqref{eq:projectors} may
therefore be recorded as
\begin{align*}
\lambda(\pv{X},\pm)&=t\pm\lVert\vct{r}\rVert,\notag\\
\vct{u}&=\vct{r}/\lVert\vct{r}\rVert,\notag\\
\pv{\Pi}(\pm)&=\tfrac12(1\pm\vct{u}\cdot\sigv),\notag\\
\pv{\Pi}(\pm)\pv{X}&=\lambda(\pv{X},\pm)\pv{\Pi}(\pm),\notag\\
\pv{X}&=\lambda(\pv{X},+)\pv{\Pi}(+)+\lambda(\pv{X},-)\pv{\Pi}(-),\notag\\
f(\pv{X})&=f\bigl(\lambda(\pv{X},+)\bigr)\pv{\Pi}(+)+f\bigl(\lambda(\pv{X},-)\bigr)\pv{\Pi}(-).
\end{align*}
The standard examples are
\begin{align*}
\exp(\pv{X})
&= \exp(t)\Bigl(\cosh(\lVert\vct{r}\rVert)+\sinh(\lVert\vct{r}\rVert)\,\vct{u}\cdot\sigv\Bigr), \\
\sqrt{\pv{X}}
&= \epsilon(+)\sqrt{a}\,\pv{\Pi}(+)+\epsilon(-)\sqrt{b}\,\pv{\Pi}(-),\notag\\
&\qquad \epsilon(+),\epsilon(-)\in\{+1,-1\}, \\
\log(\pv{X}) &= \log(a)\,\pv{\Pi}(+) + \log(b)\,\pv{\Pi}(-), \\
\pv{X}^{-1}
&= \frac{\pv{\Pi}(+)}{a} + \frac{\pv{\Pi}(-)}{b}\notag \\
&= \frac{\adjf{\pv{X}}}{\detf{\pv{X}}}
= \frac{\conj{\pv{X}}}{t^{2}-\vct{r}^{2}}, \\
\pv{X}^{n}
&= a^{n}\pv{\Pi}(+) + b^{n}\pv{\Pi}(-)\notag \\
&= \tfrac12(a^{n}+b^{n}) + \tfrac12(a^{n}-b^{n})\,\vct{u}\cdot\sigv.
\end{align*}
The scalar square roots and logarithms must be taken on specified
branches. A square root that is itself a real paravector exists when
\(a,b\geq0\), equivalently \(t\geq\lVert\vct{r}\rVert\); otherwise the
displayed roots are complex. The displayed logarithm is defined only when
\(a,b\neq0\); it is the real principal logarithm when \(a,b>0\). The condition
\(ab\neq0\) is required for the inverse. The power formula is valid for
\(n\geq0\), and also for negative integers when
\(ab\neq0\).
For two paravectors diagonalized along possibly different axes, the unit
axis of \(\pv{X}'\) is denoted by \(\vct{u}'\). The second projector pair is
the full specialization of Eq.\ \eqref{eq:projectors},
\begin{equation}
\pv{\Pi}'(\pm):=\tfrac12(1\pm\vct{u}'\cdot\sigv).
\label{eq:second-axis-projectors}
\end{equation}
Here \((\vct{u}')^2=1\). With eigenvalues \(a'\) and \(b'\) assigned
respectively to
\(\pv{\Pi}'(+)\) and \(\pv{\Pi}'(-)\),
\begin{align*}
f(\pv{X})g(\pv{X}')
&= (f(a)\pv{\Pi}(+) + f(b)\pv{\Pi}(-))\notag \\
&\quad \times (g(a')\pv{\Pi}'(+)+g(b')\pv{\Pi}'(-)),
\end{align*}
so
\begin{align*}
f(\pv{X})g(\pv{X}')
&= f(a)g(a')\,\pv{\Pi}(+)\pv{\Pi}'(+)\notag\\
&\quad+f(a)g(b')\,\pv{\Pi}(+)\pv{\Pi}'(-)\notag \\
&\quad+f(b)g(a')\,\pv{\Pi}(-)\pv{\Pi}'(+)\notag\\
&\quad+f(b)g(b')\,\pv{\Pi}(-)\pv{\Pi}'(-).
\end{align*}
The mixed projector products are
\begin{align*}
\pv{\Pi}(+)\pv{\Pi}'(+)
&= \tfrac14(1+\vct{u}\cdot\vct{u}')\notag\\
&\quad + \tfrac14\bigl(\vct{u}'+\vct{u}+i(\vct{u}\times\vct{u}')\bigr)\cdot\sigv, \\
\pv{\Pi}(+)\pv{\Pi}'(-)
&= \tfrac14(1-\vct{u}\cdot\vct{u}')\notag\\
&\quad + \tfrac14\bigl(-\vct{u}'+\vct{u}-i(\vct{u}\times\vct{u}')\bigr)\cdot\sigv, \\
\pv{\Pi}(-)\pv{\Pi}'(+)
&= \tfrac14(1-\vct{u}\cdot\vct{u}')\notag\\
&\quad + \tfrac14\bigl(\vct{u}'-\vct{u}-i(\vct{u}\times\vct{u}')\bigr)\cdot\sigv, \\
\pv{\Pi}(-)\pv{\Pi}'(-)
&= \tfrac14(1+\vct{u}\cdot\vct{u}')\notag\\
&\quad + \tfrac14\bigl(-\vct{u}'-\vct{u}+i(\vct{u}\times\vct{u}')\bigr)\cdot\sigv.
\end{align*}
As a useful example, the Cayley-type transform is itself a projector exponential:
\begin{align*}
\frac{1-\vct{r}\cdot\sigv}{1+\vct{r}\cdot\sigv}
&= \exp\!\bigl(-2\,\operatorname{arctanh}(\vct{r}\cdot\sigv)\bigr)\notag \\
&= \exp\!\bigl(-2\,\operatorname{arctanh}(\lVert\vct{r}\rVert)\bigr)\pv{\Pi}(+)\notag \\
&\quad + \exp\!\bigl(+2\,\operatorname{arctanh}(\lVert\vct{r}\rVert)\bigr)\pv{\Pi}(-).
\end{align*}
The rational expression is defined only when
\(\lVert\vct{r}\rVert\neq1\).
For a real-valued \(\operatorname{arctanh}\) and the displayed real
exponentials, the additional condition
\(\lVert\vct{r}\rVert<1\) is required; otherwise a complex branch must be
selected.

\subsubsection{Minimal Polynomial and Projector Decomposition}
The function formulas, projector identities, and exponential identities
are determined by the two-root factorization of a complex paravector, as
shown by Eqs.\ \eqref{eq:paravector-product} and
\eqref{eq:functional-calculus}:
\begin{align*}
\pv{Z} &= S + \vct{V} \cdot \sigv \in \C+\C^{3}, \\
\pv{Z}^{2} - 2 S \pv{Z} + ( S^{2} - \vct{V}^{2} ) &= 0.
\end{align*}
When \(\vct{V}^{2}\neq0\), a value \(z\) with \(z^2=\vct{V}^2\) is chosen. The two
eigenvalue values are defined by
\begin{equation}
\lambda(\pv{Z},\pm):=S\pm z.
\label{eq:complex-paravector-eigenvalues}
\end{equation}
The corresponding projector function is defined by
\begin{equation}
\pv{\Pi}(\pv{Z},\pm):=\frac12\left(1\pm
\frac{\vct{V}\cdot\sigv}{z}\right).
\label{eq:complex-paravector-projectors}
\end{equation}
Then
\begin{align*}
( \pv{Z} - \lambda(\pv{Z},-) )
( \pv{Z} - \lambda(\pv{Z},+) ) &= 0,\\
\pv{Z}&=\lambda(\pv{Z},+)\pv{\Pi}(\pv{Z},+)\notag\\
&\quad+\lambda(\pv{Z},-)\pv{\Pi}(\pv{Z},-),\\
f(\pv{Z})&=f\bigl(\lambda(\pv{Z},+)\bigr)\pv{\Pi}(\pv{Z},+)
 \notag\\
&\quad+f\bigl(\lambda(\pv{Z},-)\bigr)\pv{\Pi}(\pv{Z},-)
\end{align*}
for any scalar function defined on the two eigenvalues. The two labels are
merely exchanged when the sign of \(z\) is changed.

The case \(\vct{V}^2=0\) must be treated separately. If
\(\vct{V}=\vct{0}\), then \(\pv{Z}=S\) is scalar. If
\(\vct{V}\neq\vct{0}\), the nilpotent element is defined by
\(\pv{N}:=\vct{V}\cdot\sigv\); then \(\pv{N}^2=0\),
\(\pv{Z}=S+\pv{N}\) is not diagonalizable, and for a function analytic at
\(S\),
\begin{equation*}
f(\pv{Z})=f(S)+f'(S)\pv{N}.
\end{equation*}
This nilpotent formula is the continuous replacement for the two-projector
formula when the two roots coalesce.
