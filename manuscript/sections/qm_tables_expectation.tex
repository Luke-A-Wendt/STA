\newcommand{\bcomponentrow}[2]{\eqtext{#1} & \(\displaystyle #2\) \\}

\noindent The direct action of a complex paravector operator on a complex paravector state is recorded in Table \ref{tab:components-O-psi}.
\begin{bilineartable}[tab:components-O-psi]{\pv{\mathcal{O}}\pv{\psi}}
\bcomponentrow{scalar}{\eqtext{scalar}(\pv{\mathcal{O}})\psi_s+\eqtext{vector}(\pv{\mathcal{O}})\cdot\vct{\psi}_{\vct{v}}}
\bcomponentrow{vector}{\eqtext{scalar}(\pv{\mathcal{O}})\vct{\psi}_{\vct{v}}+\eqtext{vector}(\pv{\mathcal{O}})\psi_s+i\bigl(\eqtext{vector}(\pv{\mathcal{O}})\times\vct{\psi}_{\vct{v}}\bigr)}
\end{bilineartable}

\noindent The three basic bilinear sandwiches with a generic complex paravector \(\pv{Z}=Z_s+\vct{Z}_{\vct{v}}\cdot\sigv\) are given in Tables \ref{tab:components-psistar-Z-psi}--\ref{tab:components-psistarR-Z-psi}.
\begin{bilineartable}[tab:components-psistar-Z-psi]{\conj{\pv{\psi}}\pv{Z}\pv{\psi}}
\bcomponentrow{scalar}{\begin{aligned}[t]
&Z_s\bigl(|\psi_s|^2-\vct{\psi}_{\vct{v}}^*\cdot\vct{\psi}_{\vct{v}}\bigr)\\
&\quad+\psi_s^*(\vct{Z}_{\vct{v}}\cdot\vct{\psi}_{\vct{v}})
-\psi_s(\vct{Z}_{\vct{v}}\cdot\vct{\psi}_{\vct{v}}^*)\\
&\quad-i\,\vct{\psi}_{\vct{v}}^*\cdot(\vct{Z}_{\vct{v}}\times\vct{\psi}_{\vct{v}})
\end{aligned}}
\bcomponentrow{vector}{\begin{aligned}[t]
&Z_s\bigl(\psi_s^*\vct{\psi}_{\vct{v}}-\psi_s\vct{\psi}_{\vct{v}}^*
-i\,\vct{\psi}_{\vct{v}}^*\times\vct{\psi}_{\vct{v}}\bigr)\\
&\quad+\bigl(|\psi_s|^2+\vct{\psi}_{\vct{v}}^*\cdot\vct{\psi}_{\vct{v}}\bigr)\vct{Z}_{\vct{v}}\\
&\quad-(\vct{Z}_{\vct{v}}\cdot\vct{\psi}_{\vct{v}}^*)\vct{\psi}_{\vct{v}}
-(\vct{Z}_{\vct{v}}\cdot\vct{\psi}_{\vct{v}})\vct{\psi}_{\vct{v}}^*\\
&\quad+i\bigl(\psi_s^* \vct{Z}_{\vct{v}}\times\vct{\psi}_{\vct{v}}
+\psi_s \vct{Z}_{\vct{v}}\times\vct{\psi}_{\vct{v}}^*\bigr)
\end{aligned}}
\end{bilineartable}

\begin{bilineartable}[tab:components-psiR-Z-psi]{\herm{\pv{\psi}}\pv{Z}\pv{\psi}}
\bcomponentrow{scalar}{\begin{aligned}[t]
&Z_s\bigl(|\psi_s|^2+\vct{\psi}_{\vct{v}}^*\cdot\vct{\psi}_{\vct{v}}\bigr)\\
&\quad+\psi_s^*(\vct{Z}_{\vct{v}}\cdot\vct{\psi}_{\vct{v}})
+\psi_s(\vct{Z}_{\vct{v}}\cdot\vct{\psi}_{\vct{v}}^*)\\
&\quad+i\,\vct{\psi}_{\vct{v}}^*\cdot(\vct{Z}_{\vct{v}}\times\vct{\psi}_{\vct{v}})
\end{aligned}}
\bcomponentrow{vector}{\begin{aligned}[t]
&Z_s\bigl(\psi_s^*\vct{\psi}_{\vct{v}}+\psi_s\vct{\psi}_{\vct{v}}^*
+i\,\vct{\psi}_{\vct{v}}^*\times\vct{\psi}_{\vct{v}}\bigr)\\
&\quad+\bigl(|\psi_s|^2-\vct{\psi}_{\vct{v}}^*\cdot\vct{\psi}_{\vct{v}}\bigr)\vct{Z}_{\vct{v}}\\
&\quad+(\vct{Z}_{\vct{v}}\cdot\vct{\psi}_{\vct{v}}^*)\vct{\psi}_{\vct{v}}
+(\vct{Z}_{\vct{v}}\cdot\vct{\psi}_{\vct{v}})\vct{\psi}_{\vct{v}}^*\\
&\quad+i\bigl(\psi_s^* \vct{Z}_{\vct{v}}\times\vct{\psi}_{\vct{v}}
-\psi_s \vct{Z}_{\vct{v}}\times\vct{\psi}_{\vct{v}}^*\bigr)
\end{aligned}}
\end{bilineartable}

\begin{bilineartable}[tab:components-psistarR-Z-psi]{\cherm{\pv{\psi}}\pv{Z}\pv{\psi}}
\bcomponentrow{scalar}{Z_s\bigl(\psi_s^2-\vct{\psi}_{\vct{v}}\cdot\vct{\psi}_{\vct{v}}\bigr)}
\bcomponentrow{vector}{\begin{aligned}[t]
&\bigl(\psi_s^2+\vct{\psi}_{\vct{v}}\cdot\vct{\psi}_{\vct{v}}\bigr)\vct{Z}_{\vct{v}}\\
&\quad-2(\vct{Z}_{\vct{v}}\cdot\vct{\psi}_{\vct{v}})\vct{\psi}_{\vct{v}}
+2i\,\psi_s \vct{Z}_{\vct{v}}\times\vct{\psi}_{\vct{v}}
\end{aligned}}
\end{bilineartable}
