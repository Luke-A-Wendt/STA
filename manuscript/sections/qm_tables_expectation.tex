\newcommand{\bcomponentrow}[2]{\eqtext{#1} & \(\displaystyle #2\) \\}

\noindent The direct action of a complex paravector operator on a complex paravector state is recorded in Table \ref{tab:components-O-psi}.
\begin{bilineartable}[tab:components-O-psi]{\pv{\mathcal{O}}\pv{\psi}_{\eqtext{block}}}
\bcomponentrow{scalar}{\eqfunc{scalar}(\pv{\mathcal{O}})\,\eqfunc{scalar}(\pv{\psi}_{\eqtext{block}})+\eqfunc{vector}(\pv{\mathcal{O}})\cdot\eqfunc{vector}(\pv{\psi}_{\eqtext{block}})}
\bcomponentrow{vector}{\eqfunc{scalar}(\pv{\mathcal{O}})\,\eqfunc{vector}(\pv{\psi}_{\eqtext{block}})+\eqfunc{vector}(\pv{\mathcal{O}})\,\eqfunc{scalar}(\pv{\psi}_{\eqtext{block}})+i\bigl(\eqfunc{vector}(\pv{\mathcal{O}})\times\eqfunc{vector}(\pv{\psi}_{\eqtext{block}})\bigr)}
\end{bilineartable}

\noindent The three basic ordered two-sided products with a generic complex
paravector
\(\pv{Z}=\eqfunc{scalar}(\pv{Z})+\eqfunc{vector}(\pv{Z})\cdot\sigv\) are given in Tables
\ref{tab:components-psistar-Z-psi}--\ref{tab:components-psistarR-Z-psi}.

-(\eqfunc{vector}(\pv{Z})\cdot\eqfunc{vector}(\pv{\psi}_{\eqtext{block}}))\,\eqfunc{vector}(\pv{\psi}_{\eqtext{block}})^*\\
