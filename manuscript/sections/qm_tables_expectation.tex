% These tables are algebraic component identities.  Their vector rows are not
% identified with probability currents without an accompanying field equation.
\newenvironment{bilineartable}[2][]{%
  \begin{tableblock}
  \captionof{table}{Components of $#2$.}%
  \if\relax\detokenize{#1}\relax\else\label{#1}\fi
  \renewcommand{\arraystretch}{1.10}%
  \setlength{\tabcolsep}{2pt}%
  \small
  \begin{tabular}{@{}r@{:\quad}>{\raggedright\arraybackslash}p{0.77\columnwidth}@{}}
  \toprule
}{%
  \bottomrule
  \end{tabular}
  \end{tableblock}
}
\newcommand{\bcomponentrow}[2]{\eqtext{#1} & \(\displaystyle #2\) \\}

\noindent Table \ref{tab:components-O-psi} records the direct action of a complex paravector operator on a complex paravector state.
\begin{bilineartable}[tab:components-O-psi]{O\psi}
\bcomponentrow{scalar}{O_s\psi_s+O_{\mathbf v}\cdot\psi_{\mathbf v}}
\bcomponentrow{vector}{O_s\psi_{\mathbf v}+O_{\mathbf v}\psi_s+i(O_{\mathbf v}\times\psi_{\mathbf v})}
\end{bilineartable}

\noindent Tables \ref{tab:components-psistar-Z-psi}--\ref{tab:components-psistarR-Z-psi} give the three basic bilinear sandwiches with a generic complex paravector \(Z=Z_s+Z_{\mathbf v}\cdot\sigv\).
\begin{bilineartable}[tab:components-psistar-Z-psi]{\conj{\psi}Z\psi}
\bcomponentrow{scalar}{\begin{aligned}[t]
&Z_s\bigl(|\psi_s|^2-\psi_{\mathbf v}^*\cdot\psi_{\mathbf v}\bigr)\\
&\quad+\psi_s^*(Z_{\mathbf v}\cdot\psi_{\mathbf v})
-\psi_s(Z_{\mathbf v}\cdot\psi_{\mathbf v}^*)\\
&\quad-i\,\psi_{\mathbf v}^*\cdot(Z_{\mathbf v}\times\psi_{\mathbf v})
\end{aligned}}
\bcomponentrow{vector}{\begin{aligned}[t]
&Z_s\bigl(\psi_s^*\psi_{\mathbf v}-\psi_s\psi_{\mathbf v}^*
-i\,\psi_{\mathbf v}^*\times\psi_{\mathbf v}\bigr)\\
&\quad+\bigl(|\psi_s|^2+\psi_{\mathbf v}^*\cdot\psi_{\mathbf v}\bigr)Z_{\mathbf v}\\
&\quad-(Z_{\mathbf v}\cdot\psi_{\mathbf v}^*)\psi_{\mathbf v}
-(Z_{\mathbf v}\cdot\psi_{\mathbf v})\psi_{\mathbf v}^*\\
&\quad+i\bigl(\psi_s^* Z_{\mathbf v}\times\psi_{\mathbf v}
+\psi_s Z_{\mathbf v}\times\psi_{\mathbf v}^*\bigr)
\end{aligned}}
\end{bilineartable}

\begin{bilineartable}[tab:components-psiR-Z-psi]{\rev{\psi}Z\psi}
\bcomponentrow{scalar}{\begin{aligned}[t]
&Z_s\bigl(|\psi_s|^2+\psi_{\mathbf v}^*\cdot\psi_{\mathbf v}\bigr)\\
&\quad+\psi_s^*(Z_{\mathbf v}\cdot\psi_{\mathbf v})
+\psi_s(Z_{\mathbf v}\cdot\psi_{\mathbf v}^*)\\
&\quad+i\,\psi_{\mathbf v}^*\cdot(Z_{\mathbf v}\times\psi_{\mathbf v})
\end{aligned}}
\bcomponentrow{vector}{\begin{aligned}[t]
&Z_s\bigl(\psi_s^*\psi_{\mathbf v}+\psi_s\psi_{\mathbf v}^*
+i\,\psi_{\mathbf v}^*\times\psi_{\mathbf v}\bigr)\\
&\quad+\bigl(|\psi_s|^2-\psi_{\mathbf v}^*\cdot\psi_{\mathbf v}\bigr)Z_{\mathbf v}\\
&\quad+(Z_{\mathbf v}\cdot\psi_{\mathbf v}^*)\psi_{\mathbf v}
+(Z_{\mathbf v}\cdot\psi_{\mathbf v})\psi_{\mathbf v}^*\\
&\quad+i\bigl(\psi_s^* Z_{\mathbf v}\times\psi_{\mathbf v}
-\psi_s Z_{\mathbf v}\times\psi_{\mathbf v}^*\bigr)
\end{aligned}}
\end{bilineartable}

\begin{bilineartable}[tab:components-psistarR-Z-psi]{\crev{\psi}Z\psi}
\bcomponentrow{scalar}{Z_s\bigl(\psi_s^2-\psi_{\mathbf v}\cdot\psi_{\mathbf v}\bigr)}
\bcomponentrow{vector}{\begin{aligned}[t]
&\bigl(\psi_s^2+\psi_{\mathbf v}\cdot\psi_{\mathbf v}\bigr)Z_{\mathbf v}\\
&\quad-2(Z_{\mathbf v}\cdot\psi_{\mathbf v})\psi_{\mathbf v}
+2i\,\psi_s Z_{\mathbf v}\times\psi_{\mathbf v}
\end{aligned}}
\end{bilineartable}

\noindent For a scalar function \(S\), the same three sandwiches reduce to Tables \ref{tab:components-psistar-S-psi}--\ref{tab:components-psistarR-S-psi}.
\begin{bilineartable}[tab:components-psistar-S-psi]{\conj{\psi}S\psi}
\bcomponentrow{scalar}{S\bigl(|\psi_s|^2-\psi_{\mathbf v}^*\cdot\psi_{\mathbf v}\bigr)}
\bcomponentrow{vector}{S\bigl(\psi_s^*\psi_{\mathbf v}-\psi_s\psi_{\mathbf v}^*-i\,\psi_{\mathbf v}^*\times\psi_{\mathbf v}\bigr)}
\end{bilineartable}

\begin{bilineartable}[tab:components-psiR-S-psi]{\rev{\psi}S\psi}
\bcomponentrow{scalar}{S\bigl(|\psi_s|^2+\psi_{\mathbf v}^*\cdot\psi_{\mathbf v}\bigr)}
\bcomponentrow{vector}{S\bigl(\psi_s^*\psi_{\mathbf v}+\psi_s\psi_{\mathbf v}^*+i\,\psi_{\mathbf v}^*\times\psi_{\mathbf v}\bigr)}
\end{bilineartable}

\begin{bilineartable}[tab:components-psistarR-S-psi]{\crev{\psi}S\psi}
\bcomponentrow{scalar}{S\bigl(\psi_s^2-\psi_{\mathbf v}\cdot\psi_{\mathbf v}\bigr)}
\bcomponentrow{vector}{0}
\end{bilineartable}
