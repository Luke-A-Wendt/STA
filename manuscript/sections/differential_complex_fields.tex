\subsection{Complex Fields}\label{subsec:differential-complex-fields}
For a generic complex paravector field, write
\begin{align*}
\partial &= \partial_t+\gradv\cdot\sigv,\\
F &= (a+ib)+(\mathbf A+i\mathbf B)\cdot\sigv.
\end{align*}
Assume
\begin{align*}
a &: \R\times\R^3\to\R, &
\mathbf A &: \R\times\R^3\to\R^3,\\
b &: \R\times\R^3\to\R, &
\mathbf B &: \R\times\R^3\to\R^3,
\end{align*}
so that
\begin{equation*}
F:\R\times\R^3\to\C\oplus\C^3.
\end{equation*}
