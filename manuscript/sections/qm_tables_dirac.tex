\noindent The direct component action of \(\pv{\partial}\) and \(\conj{\pv{\partial}}\) on a complex paravector test field is recorded in Tables \ref{tab:components-partial-psi} and \ref{tab:components-partialstar-psi}.
\begin{componenttable}[tab:components-partial-psi]{\pv{\partial} \pv{\psi}}
\componentrow{scalar}{\partial_t\psi_s+\gradv\cdot\vct{\psi}_{\vct{v}}}{}
\componentrow{vector}{\partial_t\vct{\psi}_{\vct{v}}+\gradv\psi_s+i(\gradv\times\vct{\psi}_{\vct{v}})}{}
\end{componenttable}

\begin{componenttable}[tab:components-partialstar-psi]{\conj{\pv{\partial}}\pv{\psi}}
\componentrow{scalar}{\partial_t\psi_s-\gradv\cdot\vct{\psi}_{\vct{v}}}{}
\componentrow{vector}{\partial_t\vct{\psi}_{\vct{v}}-\gradv\psi_s-i(\gradv\times\vct{\psi}_{\vct{v}})}{}
\end{componenttable}

\noindent The component actions of the potential \(\pv{\Phi}\) and its conjugate on the same field are then shown in Tables \ref{tab:components-phi-psi} and \ref{tab:components-phistar-psi}.
\begin{componenttable}[tab:components-phi-psi]{\pv{\Phi}\pv{\psi}}
\componentrow{scalar}{V\psi_s+\vct{A}\cdot\vct{\psi}_{\vct{v}}}{}
\componentrow{vector}{V\vct{\psi}_{\vct{v}}+\psi_s\vct{A}+i(\vct{A}\times\vct{\psi}_{\vct{v}})}{}
\end{componenttable}

\begin{componenttable}[tab:components-phistar-psi]{\conj{\pv{\Phi}}\pv{\psi}}
\componentrow{scalar}{V\psi_s-\vct{A}\cdot\vct{\psi}_{\vct{v}}}{}
\componentrow{vector}{V\vct{\psi}_{\vct{v}}-\psi_s\vct{A}-i(\vct{A}\times\vct{\psi}_{\vct{v}})}{}
\end{componenttable}

\noindent The Dirac-block operator tables are completed by Tables
\ref{tab:components-iqf-psi} and \ref{tab:components-iqfstar-psi}, in which
the field-coupling terms \(iq\pv{F}\pv{\psi}\) and
\(iq\conj{\pv{F}}\pv{\psi}\) are recorded. In the second table,
\(\conj{\pv{F}}=-\vct{F}^{*}\cdot\sigv
=-(\vct{E}-i\vct{B})\cdot\sigv\), so no entrywise matrix conjugation is
hidden in the notation.
\begin{componenttable}[tab:components-iqf-psi]{iq\pv{F}\pv{\psi}}
\componentrow{scalar}{iq\bigl(\vct{F}\cdot\vct{\psi}_{\vct{v}}\bigr)}{}
\componentrow{vector}{iq\,\psi_s\vct{F}
-q\bigl(\vct{F}\times\vct{\psi}_{\vct{v}}\bigr)}{}
\end{componenttable}

\begin{componenttable}[tab:components-iqfstar-psi]{iq\conj{\pv{F}}\pv{\psi}}
\componentrow{scalar}{-iq\bigl(\vct{F}^{*}\cdot\vct{\psi}_{\vct{v}}\bigr)}{}
\componentrow{vector}{-iq\,\psi_s\vct{F}^{*}
+q\bigl(\vct{F}^{*}\times\vct{\psi}_{\vct{v}}\bigr)}{}
\end{componenttable}
