\noindent Tables \ref{tab:components-partial-psi} and \ref{tab:components-partialstar-psi} record the direct component action of \(\partial\) and \(\partial^{*}\) on a complex paravector test field.
\begin{componenttable}[tab:components-partial-psi]{\partial \psi}
\componentrow{scalar}{\partial_t\psi_s+\gradv\cdot\psi_{\mathbf v}}{}
\componentrow{vector}{\partial_t\psi_{\mathbf v}+\gradv\psi_s+i(\gradv\times\psi_{\mathbf v})}{}
\end{componenttable}

\begin{componenttable}[tab:components-partialstar-psi]{\partial^{*}\psi}
\componentrow{scalar}{\partial_t\psi_s-\gradv\cdot\psi_{\mathbf v}}{}
\componentrow{vector}{\partial_t\psi_{\mathbf v}-\gradv\psi_s-i(\gradv\times\psi_{\mathbf v})}{}
\end{componenttable}

\noindent Tables \ref{tab:components-phi-psi} and \ref{tab:components-phistar-psi} then show how the potential \(\Phi\) and its conjugate act on the same field in components.
\begin{componenttable}[tab:components-phi-psi]{\Phi\psi}
\componentrow{scalar}{V\psi_s+\mathbf A\cdot\psi_{\mathbf v}}{}
\componentrow{vector}{V\psi_{\mathbf v}+\psi_s\mathbf A+i(\mathbf A\times\psi_{\mathbf v})}{}
\end{componenttable}

\begin{componenttable}[tab:components-phistar-psi]{\Phi^{*}\psi}
\componentrow{scalar}{V\psi_s-\mathbf A\cdot\psi_{\mathbf v}}{}
\componentrow{vector}{V\psi_{\mathbf v}-\psi_s\mathbf A-i(\mathbf A\times\psi_{\mathbf v})}{}
\end{componenttable}

\noindent Tables \ref{tab:components-iqf-psi} and \ref{tab:components-iqfstar-psi} complete the two-component operator tables by recording the field-coupling terms \(iqF\psi\) and \(iqF^{*}\psi\).
\begin{componenttable}[tab:components-iqf-psi]{iqF\psi}
\componentrow{scalar}{iq(\mathbf F\cdot\psi_{\mathbf v})}{}
\componentrow{vector}{iq\,\psi_s \mathbf F-q(\mathbf F\times\psi_{\mathbf v})}{}
\end{componenttable}

\begin{componenttable}[tab:components-iqfstar-psi]{iqF^{*}\psi}
\componentrow{scalar}{-iq(\mathbf F^{*}\cdot\psi_{\mathbf v})}{}
\componentrow{vector}{-iq\,\psi_s \mathbf F^{*}+q(\mathbf F^{*}\times\psi_{\mathbf v})}{}
\end{componenttable}
