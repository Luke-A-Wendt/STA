\subsubsection{Space-Time Numbers}
A spacetime increment is the real paravector
\[
\dd\pv{X}=\dd t+\dd r_{1}\pv{\sigma}_{1}+\dd r_{2}\pv{\sigma}_{2}+\dd r_{3}\pv{\sigma}_{3}
=\dd t+\dd\vct{r}\cdot\sigv,
\]
with conjugate
\[
\conj{\dd\pv{X}}=\dd t-\dd r_{1}\pv{\sigma}_{1}-\dd r_{2}\pv{\sigma}_{2}-\dd r_{3}\pv{\sigma}_{3}
=\dd t-\dd\vct{r}\cdot\sigv.
\]
Here \(\dd t\in\R\) is the scalar part and \(\dd\vct{r}\in\R^{3}\) is the spatial vector part. Their recovery formulas are
\begin{equation*}
\frac{\dd\pv{X}+\conj{\dd\pv{X}}}{2}=\dd t,
\qquad
\frac{\dd\pv{X}-\conj{\dd\pv{X}}}{2}=\dd\vct{r}\cdot\sigv,
\end{equation*}
so
\begin{equation*}
\eqfunc{scalar}(\dd\pv{X})=\dd t,
\qquad
\eqfunc{vector}(\dd\pv{X})=\dd\vct{r}.
\end{equation*}
The determinant of the increment is
\begin{equation*}
\detf{\dd\pv{X}}
=\dd\pv{X}\conj{\dd\pv{X}}
=\conj{\dd\pv{X}}\dd\pv{X}
=\dd t^{2}-\dd\vct{r}^{2}
=\dd s^{2}.
\end{equation*}
The following identities are direct consequences of the difference
\(\dd t^2-\dd\vct{r}^2\):
\begin{align*}
\dd t^2>\dd\vct{r}^2,\ \dd\pv{X}\neq 0
&\eqtext{ implies }\detf{\dd\pv{X}}>0,\notag\\
\detf{\dd\pv{X}+\dd\pv{X}'}
&-\detf{\dd\pv{X}}\notag\\
&\qquad -\detf{\dd\pv{X}'}
=2(\dd t\,\dd t'-\dd\vct{r}\cdot\dd\vct{r}'),\notag\\
\detf{a\,\dd\pv{X}}
&=a^2\detf{\dd\pv{X}},
\qquad a\in\R.
\end{align*}
The commutator of two spacetime increments has zero scalar part and lies
entirely in the imaginary-vector part:
\begin{align*}
\com{\dd\pv{X}}{\dd\pv{X}'}
&=\com{\dd\vct{r}\cdot\sigv}{\dd\vct{r}'\cdot\sigv}\notag\\
&=2i(\dd\vct{r}\times\dd\vct{r}')\cdot\sigv.
\end{align*}
A path is now taken for which \(\dd t>0\) and
\(\dd t^2-\dd\vct{r}^2>0\). The first condition selects increasing time,
and the second makes the interval timelike. Such a path is conventionally
called future-directed and timelike \cite{einstein1905,minkowski1908}.
Its positive interval length satisfies
\begin{align*}
\dd s
&=\sqrt{\dd t^{2}-\dd\vct{r}^{2}}\notag\\
&=\sqrt{1-\dd\vct{r}^{2}/\dd t^{2}}\,\dd t\notag\\
&=\sqrt{1-\vct{v}^{2}}\,\dd t,
\end{align*}
where
\begin{equation*}
\vct{v}:=\frac{\dd\vct{r}}{\dd t}
=\vthree{v_{1}}{v_{2}}{v_{3}}\in\R^{3},
\qquad
\gamma:=\frac{\dd t}{\dd s}=\frac{1}{\sqrt{1-\vct{v}^{2}}}.
\end{equation*}
The factor \(\gamma\), which compares coordinate time with proper time, is
conventionally called the Lorentz factor.
For a real non-null increment, \(\herm{\dd\pv{X}}=\dd\pv{X}\), so the inverse exists and
\begin{align*}
(\dd\pv{X})^{-1}
&=\frac{\adjf{\dd\pv{X}}}{\detf{\dd\pv{X}}}\notag\\
&=\frac{\conj{\dd\pv{X}}}{\detf{\dd\pv{X}}}\notag\\
&=\frac{\conj{\dd\pv{X}}}{\dd\pv{X}\conj{\dd\pv{X}}}\notag\\
&=\frac{\conj{\dd\pv{X}}}{\dd s^{2}}.
\end{align*}
An increment with \(\dd s^{2}=0\) is called null below. Such increments are
singled out algebraically because the inverse breaks down at exactly that
value.

\subsubsection{Interval Classes and Proper Time}
For a spacetime increment \(\dd\pv{X}=\dd t+\dd \vct{r}\cdot\sigv\), the interval is
\begin{equation*}
\detf{\dd\pv{X}}=\dd\pv{X}\,\conj{\dd\pv{X}}=\dd t^2-\dd \vct{r}^2=\dd s^2.
\end{equation*}
The sign of this displayed quantity defines the three standard interval
classes \cite{lorentz1904,einstein1905,minkowski1908}:
\begin{align*}
\dd s^2>0 &\quad\eqtext{implies a timelike interval},\notag \\
\dd s^2=0,\ \dd\pv{X}\neq0
&\quad\eqtext{implies a null interval},\notag \\
\dd s^2<0 &\quad\eqtext{implies a spacelike interval}.
\end{align*}
If \(\detf{\dd\pv{X}}>0\), the interval is timelike. The time read by a
clock moving along the path is called proper time. In the units used here,
it is the accumulated positive interval length \(s\):
\begin{align*}
s=\int \dd s
&=\int \sqrt{1-\vct{v}^2}\,\dd t\notag\\
&=\int \frac{1}{\gamma}\,\dd t.
\end{align*}
The difference between \(\dd t\) and the moving clock reading \(\dd s\)
is the time-dilation relation. In the International System of Units (SI),
the interval has length units, and the corresponding clock-time increment is
\begin{equation*}
\frac{\dd s}{c}.
\end{equation*}

If \(\detf{\dd\pv{X}}=0\) with \(\dd\pv{X}\neq0\), the interval is null or
lightlike. Then
\begin{equation*}
\dd t^2=\dd\vct{r}^2
\qquad\eqtext{implies}\qquad
\dd t=\pm \|\dd\vct{r}\|,
\end{equation*}
and \((\dd\pv{X})^{-1}\) does not exist.

If \(\detf{\dd\pv{X}}<0\), the interval is spacelike.

Thus the interval classification, the moving-clock formula, and the failure
of \((\dd\pv{X})^{-1}\) for a null increment all follow from the one
algebraic identity
\(\dd\pv{X}\,\conj{\dd\pv{X}}=\dd t^2-\dd\vct{r}^2\).

\subsubsection{Four-Velocity}
A particle path through space and time is called its worldline. For a
timelike worldline, division of its increment by the corresponding proper
time gives
\begin{align*}
\frac{\dd\pv{X}}{\dd s}
&=\frac{\dd t}{\dd s}+\frac{\dd\vct{r}}{\dd s}\cdot\sigv\notag\\
&=\frac{\dd t+\sigv\cdot\dd\vct{r}}{\sqrt{\dd t^2-\dd\vct{r}^2}}\notag\\
&=\frac{\dd t/\dd t+\sigv\cdot(\dd\vct{r}/\dd t)}{\sqrt{(\dd t/\dd t)^2-(\dd\vct{r}/\dd t)^2}}\notag\\
&=\frac{1+\vct{v}\cdot\sigv}{\sqrt{1-\vct{v}^2}}\notag\\
&=\gamma(1+\vct{v}\cdot\sigv).
\end{align*}
Its scalar and vector parts are therefore
\begin{equation*}
\eqfunc{scalar}\!\left(\frac{\dd\pv{X}}{\dd s}\right)=\gamma,
\qquad
\eqfunc{vector}\!\left(\frac{\dd\pv{X}}{\dd s}\right)=\gamma\vct{v}.
\end{equation*}
Its determinant is unity:
\begin{align*}
\detf{\frac{\dd\pv{X}}{\dd s}}
&=\gamma^2(1+\vct{v}\cdot\sigv)(1-\vct{v}\cdot\sigv)\notag\\
&=\gamma^2(1-\vct{v}^2)\notag\\
&=1.
\end{align*}
Also,
\begin{equation*}
\frac{\dd\pv{X}}{\dd t}
=\frac{\dd\pv{X}}{\dd s}\frac{\dd s}{\dd t}
=\frac{1}{\gamma}\frac{\dd\pv{X}}{\dd s}.
\end{equation*}
This determinant-one rate of change is called the four-velocity paravector
and is defined by
\begin{align*}
\pv{U}&=\frac{\dd\pv{X}}{\dd s}\notag\\
&=\gamma+\gamma\vct{v}\cdot\sigv,\\
\eqfunc{scalar}(\pv{U})&=\gamma,\qquad
\eqfunc{vector}(\pv{U})=\gamma\vct{v}.
\end{align*}
Since \(\dd s^2=\dd\pv{X}\conj{\dd\pv{X}}\) with \(\dd s>0\),
\begin{align*}
\frac{\conj{\dd\pv{X}}}{\dd s}
&=(\dd\pv{X})^{-1}\dd s\notag\\
&=\left(\frac{\dd\pv{X}}{\dd s}\right)^{-1},
\end{align*}
so \(\pv{U}^{-1}=\pv{U}^*\) and
\begin{equation*}
\detf{\pv{U}}=\pv{U}\pv{U}^*=1.
\end{equation*}
Differentiation of the determinant-one condition gives
\begin{align*}
\frac{\dd}{\dd s}\detf{\pv{U}}
&=\frac{\dd\pv{U}}{\dd s}\pv{U}^*+\pv{U}\left(\frac{\dd\pv{U}}{\dd s}\right)^*\notag\\
&=\left(\frac{\dd\pv{U}}{\dd s}\pv{U}^*\right)+\left(\frac{\dd\pv{U}}{\dd s}\pv{U}^*\right)^{*\Hop},
\end{align*}
hence
\begin{equation*}
\eqfunc{scalar}\!\left(\frac{\dd\pv{U}}{\dd s}\pv{U}^*\right)
=\eqfunc{scalar}\!\left(\pv{U}^*\frac{\dd\pv{U}}{\dd s}\right)
=0.
\end{equation*}
The determinant identity is recovered as
\begin{align*}
\detf{\dd\pv{X}}
&=\dd\pv{X}\conj{\dd\pv{X}}\notag\\
&=\left(\frac{\dd\pv{X}}{\dd s}\dd s\right)\left(\left(\frac{\dd\pv{X}}{\dd s}\right)^*\dd s\right)\notag\\
&=\pv{U}\pv{U}^*\,\dd s^2\notag\\
&=\detf{\pv{U}}\dd s^2\notag\\
&=\dd s^2.
\end{align*}
The acceleration formulas below are based on this normalization and on the
result that the displayed scalar part vanishes.

\subsubsection{Four-Acceleration}
The rate of change of the three-velocity with respect to the chosen time
coordinate is called the coordinate acceleration and is defined by
\begin{equation*}
\vct{a}:=\frac{\dd\vct{v}}{\dd t}
=\vthree{a_1}{a_2}{a_3}\in\R^3.
\end{equation*}
With \(\gamma=(1-\vct{v}^2)^{-1/2}\), the following relations hold:
\begin{align*}
\frac{\dd\gamma}{\dd t}&=\gamma^3(\vct{v}\cdot\vct{a}),\\
\frac{\dd\gamma}{\dd s}&=\gamma^4(\vct{v}\cdot\vct{a}).
\end{align*}
The scalar derivative with respect to the three velocity components is
\[
\frac{\partial}{\partial\vct{v}}\!\left(-\frac{1}{\gamma}\right)
=\gamma\vct{v}.
\]
The following derivatives are obtained by direct differentiation of the vector product \(\gamma\vct{v}\):
\begin{align*}
\frac{\dd(\gamma\vct{v})}{\dd t}
&=\gamma\vct{a}+\frac{\dd\gamma}{\dd t}\vct{v}\notag\\
&=\gamma\vct{a}+\gamma^3(\vct{v}\cdot\vct{a})\vct{v},\\
\frac{\dd(\gamma\vct{v})}{\dd s}
&=\frac{\dd(\gamma\vct{v})}{\dd t}\frac{\dd t}{\dd s}\notag\\
&=\gamma^2\vct{a}+\gamma^4(\vct{v}\cdot\vct{a})\vct{v}.
\end{align*}

The scalar and vector parts of four-velocity are
\(\eqfunc{scalar}(\pv{U})=\gamma\) and
\(\eqfunc{vector}(\pv{U})=\gamma\vct{v}\), as obtained above.
The rate of change of four-velocity with respect to proper time is called
four-acceleration. It is therefore
\begin{align*}
\pv{A}
&=\frac{\dd\pv{U}}{\dd s}\notag\\
&=\frac{\dd\gamma}{\dd s}+\frac{\dd(\gamma\vct{v})}{\dd s}\cdot\sigv\notag\\
&=\gamma^4(\vct{v}\cdot\vct{a})
+\Bigl(\gamma^4(\vct{v}\cdot\vct{a})\vct{v}+\gamma^2\vct{a}\Bigr)\cdot\sigv.
\end{align*}
Equivalently,
\begin{align*}
\eqfunc{scalar}(\pv{A})
&=\gamma^4(\vct{v}\cdot\vct{a}),\\
\eqfunc{vector}(\pv{A})
&=\gamma^4(\vct{v}\cdot\vct{a})\vct{v}+\gamma^2\vct{a}.
\end{align*}

For \(\vct{v}\neq 0\), the unit direction is defined by
\(\vct{u}:=\vct{v}/\|\vct{v}\|\), with \(\vct{u}^2=1\). The following
components are then obtained from the axis functions in
Eqs.\ \eqref{eq:par-function} and \eqref{eq:prp-function}:
\begin{align*}
\eqfunc{par}(\vct{a})&=(\vct{a}\cdot\vct{u})\vct{u},\\
\eqfunc{prp}(\vct{a})&=\vct{a}-\eqfunc{par}(\vct{a}).
\end{align*}
Then
\begin{align*}
\vct{a}^2&=\eqfunc{par}(\vct{a})^2+\eqfunc{prp}(\vct{a})^2,\\
(\vct{v}\cdot\vct{a})^2&=\vct{v}^2\,\eqfunc{par}(\vct{a})^2,
\end{align*}
so the proper-acceleration determinant is
\begin{align*}
\detf{\pv{A}}
&=-\gamma^4\Bigl(\vct{a}^2+\gamma^2(\vct{v}\cdot\vct{a})^2\Bigr)\notag\\
&=-\gamma^4\Bigl(\eqfunc{prp}(\vct{a})^2+\gamma^2\eqfunc{par}(\vct{a})^2\Bigr).
\end{align*}
The determinant is negative unless the acceleration vanishes, so the
four-acceleration is spacelike. Its positive magnitude
\(\sqrt{-\detf{\pv{A}}}\) is unchanged by a Lorentz transformation. It is
called proper acceleration because it equals the acceleration measured at
that event in the momentarily comoving frame, where the particle velocity is
zero.

The products with \(\pv{U}\) are
\begin{align*}
\pv{U}\pv{A}
&=\bigl(
\eqfunc{scalar}(\pv{U})\eqfunc{scalar}(\pv{A})
+\eqfunc{vector}(\pv{U})\cdot\eqfunc{vector}(\pv{A})
\bigr)\notag\\
&\quad+\bigl(
\eqfunc{scalar}(\pv{U})\eqfunc{vector}(\pv{A})
+\eqfunc{scalar}(\pv{A})\eqfunc{vector}(\pv{U})\notag\\
&\qquad\qquad
+i\bigl(\eqfunc{vector}(\pv{U})\times\eqfunc{vector}(\pv{A})\bigr)
\bigr)\cdot\sigv,\\
\pv{U}^*\pv{A}
&=\bigl(
\eqfunc{scalar}(\pv{U})\eqfunc{scalar}(\pv{A})
-\eqfunc{vector}(\pv{U})\cdot\eqfunc{vector}(\pv{A})
\bigr)\notag\\
&\quad+\bigl(
\eqfunc{scalar}(\pv{U})\eqfunc{vector}(\pv{A})
-\eqfunc{scalar}(\pv{A})\eqfunc{vector}(\pv{U})\notag\\
&\qquad\qquad
-i\bigl(\eqfunc{vector}(\pv{U})\times\eqfunc{vector}(\pv{A})\bigr)
\bigr)\cdot\sigv.
\end{align*}
Their component forms are
\begin{align*}
\pv{U}\pv{A}
&=2\gamma^5(\vct{v}\cdot\vct{a})\notag\\
&\quad+\bigl(\gamma^3\vct{a}+2\gamma^5(\vct{v}\cdot\vct{a})\vct{v}\bigr)\cdot\sigv\notag\\
&\quad+i\,\gamma^3(\vct{v}\times\vct{a})\cdot\sigv,\\
\pv{U}^*\pv{A}
&=\gamma^3\vct{a}\cdot\sigv-i\,\gamma^3(\vct{v}\times\vct{a})\cdot\sigv.
\end{align*}
In particular,
\begin{equation*}
\eqfunc{scalar}(\pv{U}^*\pv{A})=0.
\end{equation*}
The vanishing scalar part is the precise meaning of orthogonality between
four-velocity and four-acceleration in this notation. The scalar and vector
parts of \(\pv{U}\) are recorded in Table
\ref{tab:components-proper-velocity}, while the corresponding parts of
\(\pv{U}\pv{A}\) and \(\pv{U}^{*}\pv{A}\) are compared in Tables
\ref{tab:components-ua} and \ref{tab:components-ustara}.

\subsubsection{Four-Momentum}
For a massive particle with rest mass \(m>0\), the energy--momentum paravector is produced by multiplication of four-velocity by \(m\):
\begin{equation*}
m\pv{U}=E+\vct{P}\cdot\sigv.
\end{equation*}
Its scalar part is the relativistic energy,
\begin{align*}
E
&:=\eqfunc{scalar}(m\pv{U})\notag\\
&=m\gamma.
\end{align*}
If \(\vct{v}=0\), then
\begin{equation*}
E=m
\qquad \text{rest energy}
\qquad (=mc^{2}\text{ in SI units}).
\end{equation*}
The kinetic energy is therefore \(E-m\).

Its vector part is the relativistic momentum,
\begin{align*}
\vct{P}
&:=\eqfunc{vector}(m\pv{U})\notag\\
&=m\gamma\vct{v}.
\end{align*}

Because \(\detf{\pv{U}}=1\), the following restriction on the allowed energy
and momentum is obtained immediately:
\begin{align*}
\detf{m\pv{U}}
&=m^{2}\notag\\
&=(E+\vct{P}\cdot\sigv)(E-\vct{P}\cdot\sigv)\notag\\
&=E^{2}-\vct{P}^{2}.
\end{align*}
Hence
\begin{equation*}
E^{2}=\vct{P}^{2}+m^{2}.
\end{equation*}
The set of energy--momentum values satisfying this equation is
conventionally called the mass shell \cite{dirac1928,greiner2000}. For
massless motion it reduces to the null branch
\(E^{2}=\vct{P}^{2}\). That branch must be introduced directly as a null
energy--momentum paravector: the equation
\(m\pv{U}=E+\vct{P}\cdot\sigv\) applies only to massive timelike motion
because \(\dd s=0\) and \(\pv{U}=\dd\pv{X}/\dd s\) is undefined on a null
worldline.

\subsubsection{Flat Paravector Coordinate Map}
The same flat event paravector may be described by functions of another
time label \(t\) and three position labels \(\vct{r}\). The scalar and
vector functions are taken to have the domains
\begin{align*}
f&:\R\times\R^3\to\R,\\
\vct{g}&:\R\times\R^3\to\R^3.
\end{align*}
Their outputs are used as the scalar and vector components of a real
paravector:
\begin{equation}
\pv{X}:=f(t,\vct{r})+\vct{g}(t,\vct{r})\cdot\sigv.
\label{eq:flat-paravector-coordinate-map}
\end{equation}
Thus \((t,\vct{r})\) supplies labels, while
\((f,\vct{g})\) supplies the components inserted into the already defined
flat interval. The numerical coefficients of that interval need not remain
constant when nonlinear labels are used. The differential and its conjugate
are
\begin{equation*}
\dd\pv{X}=\dd f+\dd\vct{g}\cdot\sigv,
\qquad
\conj{\dd\pv{X}}=\dd f-\dd\vct{g}\cdot\sigv.
\end{equation*}
Their product is
\begin{align*}
\detf{\dd\pv{X}}
&=\dd\pv{X}\,\conj{\dd\pv{X}}\notag\\
&=(\dd f+\dd\vct{g}\cdot\sigv)(\dd f-\dd\vct{g}\cdot\sigv)\notag\\
&=\dd f^2-\dd\vct{g}^2.
\end{align*}
Along a future-directed timelike path, the conditions \(\dd f>0\) and
\(\dd f^2>\dd\vct{g}^2\) are assumed. Then
\begin{align*}
\dd s
&=\sqrt{\dd f^2-\dd\vct{g}^2}\notag\\
&=\sqrt{1-\left(\frac{\dd\vct{g}}{\dd f}\right)^2}\,\dd f.
\end{align*}
Within this subsection, the bare symbols \(\vct{v}\), \(\gamma\), and
\(\vct{a}\) are defined relative to the components \((f,\vct{g})\).
They reduce to the earlier definitions when \(f=t\) and \(\vct{g}=\vct{r}\).
The velocity and Lorentz factor are
\begin{align*}
\vct{v}
&:=\frac{\dd\vct{g}}{\dd f},\\
\gamma
&:=\frac{\dd f}{\dd s}\notag\\
&=\frac{\dd f}{\sqrt{\dd f^2-\dd\vct{g}^2}}\notag\\
&=\frac{1}{\sqrt{1-\vct{v}^2}}.
\end{align*}
The corresponding four-velocity is
\begin{align*}
\pv{U}
&=\frac{\dd\pv{X}}{\dd s}\notag\\
&=\frac{\dd f+\dd\vct{g}\cdot\sigv}{\sqrt{\dd f^2-\dd\vct{g}^2}}\notag\\
&=\frac{1+\left(\frac{\dd\vct{g}}{\dd f}\right)\cdot\sigv}{\sqrt{1-\left(\frac{\dd\vct{g}}{\dd f}\right)^2}}\notag\\
&=\frac{1+\vct{v}\cdot\sigv}{\sqrt{1-\vct{v}^2}}\notag\\
&=\gamma(1+\vct{v}\cdot\sigv),
\end{align*}
where the division by \(\dd f\) is valid because \(\dd f>0\).
Thus \(\vct{v}^2<1\). For a massive particle, the energy--momentum relation remains unchanged:
\begin{equation*}
m\pv{U}=E+\vct{P}\cdot\sigv.
\end{equation*}
The coordinate acceleration relative to \(f\) is defined, in direct analogy
with \(\vct{a}=\dd\vct{v}/\dd t\) above, by
\begin{align*}
\vct{a}
&:=\frac{\dd\vct{v}}{\dd f}\notag\\
&=\frac{\dd^2\vct{g}}{\dd f^2}.
\end{align*}
Because \(\dd/\dd s=\gamma\,\dd/\dd f\), the four-acceleration has the same form as before:
\begin{equation*}
\frac{\dd\pv{U}}{\dd s}
=\gamma^4(\vct{v}\cdot\vct{a})
+\bigl(\gamma^4(\vct{v}\cdot\vct{a})\vct{v}+\gamma^2\vct{a}\bigr)\cdot\sigv.
\end{equation*}
The commutator of four-velocity and four-acceleration is
\begin{align*}
\com{\pv{U}}{\dfrac{\dd\pv{U}}{\dd s}}
&=\com{\pv{U}}{\dfrac{\dd\pv{U}}{\dd f}}\frac{\dd f}{\dd s}\notag\\
&=\gamma\com{\pv{U}}{\dfrac{\dd\pv{U}}{\dd f}}\notag\\
&=2\gamma^3(\vct{v}\times\vct{a})\cdot(i\sigv).
\end{align*}
For a massive particle on which no force acts, four-acceleration vanishes in
the flat components \((f,\vct{g})\):
\begin{equation*}
m\frac{\dd\pv{U}}{\dd s}=0,
\end{equation*}
which is equivalently the straight-worldline equation
\begin{equation*}
\frac{\dd^2\pv{X}}{\dd s^2}=0.
\end{equation*}
After division by the nonzero factor \(\gamma\), its scalar and vector parts are
\begin{align*}
\gamma^3(\vct{v}\cdot\vct{a})&=0,\\
\gamma^3(\vct{v}\cdot\vct{a})\vct{v}+\gamma\vct{a}&=0.
\end{align*}
For \(\vct{v}^2<1\), one has \(\gamma>0\). The relation
\(\vct{v}\cdot\vct{a}=0\) is therefore obtained from the scalar part, and
\(\vct{a}=0\) is then obtained from the vector part. Hence
\begin{equation*}
\frac{\dd^2\vct{g}}{\dd f^2}=0,
\end{equation*}
and along such a force-free worldline
\begin{equation*}
\vct{g}=\vct{c}f+\vct{c}',
\qquad
\vct{v}=\frac{\dd\vct{g}}{\dd f}=\vct{c}.
\end{equation*}
Thus motion without force is linear in \((f,\vct{g})\), with constant
velocity. If nonlinear coordinate labels are used for this same straight
path, differentiating the coordinate map twice produces extra terms from
the second derivatives of \(f\) and \(\vct{g}\). Such terms are
conventionally called connection terms. They can be caused entirely by the
coordinate labels and therefore do not by themselves describe curved
space-time. Curvature would require a change in the interval geometry that
cannot be removed by any coordinate relabeling throughout a region. A path
whose force-free equation includes the required
connection terms is conventionally called a geodesic
\cite{doran2003}. Only a straight path in the fixed flat components has
been derived here; no equation for a curved space-time has been assumed or
proved.

\subsubsection{Lagrangian Mechanics and Conserved Quantities}
For a particle trajectory written in terms of time, position, and velocity,
the action is defined by
\begin{equation*}
\int \mathcal L(t,\vct{r},\vct{v})\,\dd t
=:\ \eqtext{action}.
\end{equation*}
Its integrand \(\mathcal L\) is called the Lagrangian. Requiring the
first-order change of the action to
vanish for fixed endpoints gives the Euler--Lagrange equation
\cite{doran2003}:
\begin{align*}
\gradv\mathcal L
-\frac{\dd}{\dd t}
\left(\frac{\partial\mathcal L}{\partial\vct{v}}\right)&=0,\\
\frac{\dd}{\dd t}
\left(\frac{\partial\mathcal L}{\partial\vct{v}}\right)
&=\gradv\mathcal L.
\end{align*}
Two constant quantities follow directly when the Lagrangian lacks the
corresponding dependence. Canonical momentum is defined by
\begin{equation*}
\frac{\partial\mathcal L}{\partial\vct{v}}
=:\ \eqtext{canonical momentum}.
\end{equation*}
If the Lagrangian has no position dependence,
so that \(\gradv\mathcal L=0\), then
\begin{equation*}
\frac{\dd}{\dd t}
\left(\frac{\partial\mathcal L}{\partial\vct{v}}\right)=0,
\end{equation*}
and the canonical momentum is constant along the path.

The scalar combination is defined by
\begin{equation}
\begin{aligned}
\mathcal H
&:=\vct{v}\cdot\frac{\partial\mathcal L}{\partial\vct{v}}
-\mathcal L\\
&=\left(\vct{v}\cdot\frac{\partial}{\partial\vct{v}}-1\right)
\mathcal L
\end{aligned}
\label{eq:hamiltonian-definition}
\end{equation}
and is called the Hamiltonian. Its time derivative is
\begin{align*}
\frac{\dd\mathcal H}{\dd t}
&=\frac{\dd}{\dd t}
\left(\frac{\partial\mathcal L}{\partial\vct{v}}\right)\cdot\vct{v}
+\frac{\partial\mathcal L}{\partial\vct{v}}\cdot
\frac{\dd\vct{v}}{\dd t}
-\frac{\dd\mathcal L}{\dd t}\notag\\
&=(\gradv\mathcal L)\cdot\vct{v}
+\frac{\partial\mathcal L}{\partial\vct{v}}\cdot
\frac{\dd\vct{v}}{\dd t}
-\frac{\dd\mathcal L}{\dd t}\notag\\
&=(\gradv\mathcal L)\cdot\vct{v}
+\frac{\partial\mathcal L}{\partial\vct{v}}\cdot
\frac{\dd\vct{v}}{\dd t}\notag\\
&\qquad-\left(
\partial_t\mathcal L
+(\gradv\mathcal L)\cdot\vct{v}
+\frac{\partial\mathcal L}{\partial\vct{v}}\cdot
\frac{\dd\vct{v}}{\dd t}
\right)\notag\\
&=-\partial_t\mathcal L.
\end{align*}
Therefore, if \(\partial_t\mathcal L=0\), then
\begin{equation*}
\frac{\dd\mathcal H}{\dd t}=0,
\end{equation*}
so the Hamiltonian is constant along the path. These identities are applied
directly to the free-particle example and to the electromagnetic example in
which the potential is added to the action.

\subsubsection{Action of a Free Particle}
The free-particle action is negative mass times the accumulated proper time
along the worldline. With \(\pv{U}=\dd\pv{X}/\dd s\) and
\(\gamma=\dd t/\dd s\),
\begin{align*}
\int \mathcal L\,\dd t
&=-m\int \sqrt{\detf{\dd\pv{X}}}\notag\\
&=-m\int \dd s\notag\\
&=-m\int \sqrt{\detf{\frac{\dd\pv{X}}{\dd t}}}\,\dd t\notag\\
&=-m\int \frac{\dd s}{\dd t}\,\dd t\notag\\
&=-m\int \sqrt{\detf{\frac{\pv{U}}{\gamma}}}\,\dd t\notag\\
&=-m\int \frac{1}{\gamma}\,\dd t\notag\\
&=-m\int \sqrt{1-\vct{v}^2}\,\dd t.
\end{align*}
Therefore
\begin{equation*}
\mathcal L=-\frac{m}{\gamma}.
\end{equation*}
The canonical momentum is obtained by differentiation with respect to velocity:
\begin{align*}
\frac{\partial\mathcal L}{\partial\vct{v}}
&=m\gamma\vct{v}\notag\\
&=\vct{P},\\
\left(\frac{\partial}{\partial\vct{v}}\cdot\sigv\right)\mathcal L
&=\vct{P}\cdot\sigv.
\end{align*}
The Hamiltonian is
\begin{align*}
\mathcal H
&=\frac{\partial\mathcal L}{\partial\vct{v}}\cdot\vct{v}
-\mathcal L\notag\\
&=m\gamma\vct{v}^2+\frac{m}{\gamma}\notag\\
&=m\gamma\left(\vct{v}^2+\frac{1}{\gamma^2}\right)\notag\\
&=m\gamma\notag\\
&=E.
\end{align*}
Because the free Lagrangian has no explicit time or position dependence,
\begin{equation*}
\partial_t\mathcal L=0,
\qquad
\gradv\mathcal L=0,
\end{equation*}
so the conserved quantities are the relativistic energy \(E=m\gamma\) and momentum \(\vct{P}=m\gamma\vct{v}\).

\subsubsection{Low-Velocity Approximation}
When the speed is small compared with \(c=1\), the motion is conventionally
called non-relativistic. The free Lagrangian is then expanded directly:
\begin{align*}
\mathcal L
&=-\frac{m}{\gamma}\notag\\
&=-m+\frac12 m\vct{v}^2+O(\vct{v}^4).
\end{align*}
The canonical momentum is then reduced to
\begin{align*}
\frac{\partial\mathcal L}{\partial\vct{v}}
&=\frac{\partial}{\partial \vct{v}}\left(\frac12 m\vct{v}^2+O(\vct{v}^4)\right)\notag\\
&=m\vct{v}+O(\vct{v}^3)\notag\\
&\approx m\vct{v}.
\end{align*}
The total energy is the kinetic term plus the rest-mass term:
\begin{align*}
\mathcal H
&=\bigl(m\vct{v}+O(\vct{v}^3)\bigr)\cdot\vct{v}
+m-\frac12m\vct{v}^2-O(\vct{v}^4)\notag\\
&=m+\frac12m\vct{v}^2+O(\vct{v}^4)\notag\\
&\approx m+\frac12m\vct{v}^2.
\end{align*}
Equivalently, the low-velocity series is
\begin{align*}
\mathcal L&=-m+\frac12 m\vct{v}^2+O(\vct{v}^4),\notag \\
\vct{P}&=m\vct{v}+O(\vct{v}^{3}),\notag \\
\mathcal H&=m+\frac12 m\vct{v}^{2}+O(\vct{v}^{4}).
\end{align*}
The Newtonian free-particle kinetic term together with the additive
rest-energy constant is thereby obtained as the low-velocity approximation
to the relativistic action.
